📡 POSTMESSAGE COMMUNICATION ▼

LISTEN FOR WALLET RESPONSE VIA POSTMESSAGE:

// Add this to your app to listen for wallet responses
window.addEventListener('message', function(event) {
  // Only accept messages from Fixorium Wallet
  if (event.origin !== 'https://wallet.fixorium.com.pk') return;
  
  try {
    const data = typeof event.data === 'string' ? JSON.parse(event.data) : event.data;
    
    if (data.type === 'CONNECTION_APPROVED') {
      console.log('✅ Wallet connected:', data.payload.publicKey);
      // Update your UI with the connection
    }
    
    if (data.type === 'CONNECTION_REJECTED') {
      console.log('❌ Connection rejected');
    }
  } catch (error) {
    // Not a wallet message
  }
});

JAVASCRIPT - CONNECT WALLET:

async function connectWallet() {
  const requestId = 'conn_' + Date.now();
  const params = new URLSearchParams();
  params.append('requestId', requestId);
  params.append('message', 'Connect to My DApp');
  params.append('appName', 'My DApp');
  params.append('callbackUrl', window.location.origin + '/callback');

  const url = 'https://wallet.fixorium.com.pk/sign?' + params.toString();
  window.open(url, '_blank');

  // Handle callback at /callback endpoint
  // Also listen for postMessage
  window.addEventListener('message', function(event) {
    if (event.origin === 'https://wallet.fixorium.com.pk') {
      const data = JSON.parse(event.data);
      if (data.type === 'CONNECTION_APPROVED') {
        console.log('Connected:', data.payload.publicKey);
      }
    }
  });
}

JAVASCRIPT - SIGN TRANSACTION:

async function signTransaction(tx) {
  const requestId = 'tx_' + Date.now();
  const txBase64 = btoa(JSON.stringify(tx));
  const params = new URLSearchParams();
  params.append('requestId', requestId);
  params.append('transaction', txBase64);
  params.append('appName', 'My DApp');
  params.append('callbackUrl', window.location.origin + '/callback');

  const url = 'https://wallet.fixorium.com.pk/sign?' + params.toString();
  window.open(url, '_blank');
}

REACT COMPONENT:

const ConnectWallet = () => {
  const [isConnected, setIsConnected] = useState(false);
  const [publicKey, setPublicKey] = useState(null);

  useEffect(() => {
    const handler = (event) => {
      if (event.origin === 'https://wallet.fixorium.com.pk') {
        const data = JSON.parse(event.data);
        if (data.type === 'CONNECTION_APPROVED') {
          setPublicKey(data.payload.publicKey);
          setIsConnected(true);
        }
      }
    };
    window.addEventListener('message', handler);
    return () => window.removeEventListener('message', handler);
  }, []);

  const connect = () => {
    const id = 'conn_' + Date.now();
    const url = 'https://wallet.fixorium.com.pk/sign?' +
      'requestId=' + id +
      '&message=Connect+to+MyApp' +
      '&appName=MyApp' +
      '&callbackUrl=' + encodeURIComponent(window.location.origin + '/callback');
    window.open(url, '_blank');
  };

  return (
    <button onClick={connect}>
      {isConnected ? 'Connected: ' + publicKey.slice(0, 8) : 'Connect Wallet'}
    </button>
  );
};
