STEP 4: REACT INTEGRATION

import { useState } from 'react';

const BASE_URL = 'https://fixorium.com.pk';
const API_KEY = process.env.REACT_APP_MAX_API_KEY;

export default function SwapWidget() {
    const [quote, setQuote] = useState(null);

    const getQuote = async () => {
        const res = await fetch(`${BASE_URL}/api/sol-router/quote?from=SOL&to=USDC&amount=1`, {
            headers: { 'X-API-Key': API_KEY }
        });
        const data = await res.json();
        setQuote(data);
    };

    return (
        <button onClick={getQuote}>GET QUOTE</button>
        {quote && <div>{quote.best_route.output} USDC</div>}
    );
}
